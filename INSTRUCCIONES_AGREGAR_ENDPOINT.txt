@api.route('/verify-teacher-email', methods=['POST'])
def verify_teacher_email():
    """
    Verificar si un correo personal corresponde a un docente registrado.
    Usado para autenticación con Google Sign-In.
    
    Request Body:
    {
        "correo_personal": "docente@uceva.edu.co"
    }
    
    Response Success (200):
    {
        "success": true,
        "found": true,
        "docente": {
            "id_docente": 1,
            "identificacion": "1234567890",
            "nombre_docente": "GIOVANNA GIL ISAZIGA",
            "correo": "giovanna@uceva.edu.co",
            "correo_personal": "giovanna.gil@uceva.edu.co",
            "telefono": "3123456789",
            "celular": "3123456789",
            "dedicacion": "HORA CATEDRA",
            "vinculacion": "Catedrático",
            "docentes_en_estado": "Activo",
            "periodo_semestral": "2025-2",
            "nivel_academico": "Maestría"
        }
    }
    
    Response Error (404):
    {
        "success": false,
        "found": false,
        "message": "No autorizado"
    }
    """
    try:
        # Obtener funciones importadas (patrón del servidor)
        imports = get_imports()
        if not imports:
            return jsonify({
                'success': False,
                'message': 'Error cargando módulos del servidor'
            }), 500
        
        # Extraer datos del request
        data = request.get_json()
        
        if not data or 'correo_personal' not in data:
            return jsonify({
                'success': False,
                'message': 'correo_personal es requerido'
            }), 400
        
        correo_personal = data['correo_personal'].strip().lower()
        
        # Consultar la tabla docentes
        conn = None
        cursor = None
        try:
            conn = imports['get_db_connection']()
            cursor = conn.cursor(cursor_factory=imports['RealDictCursor'])
            
            # Buscar docente por correo personal (case-insensitive)
            cursor.execute("""
                SELECT 
                    id_docente,
                    identificacion,
                    nombre_docente,
                    correo,
                    "Correo personal" as correo_personal,
                    telefono,
                    celular,
                    dedicacion,
                    vinculacion,
                    docentes_en_estado,
                    periodo_semestral,
                    nivel_academico
                FROM docentes 
                WHERE LOWER("Correo personal") = %s
            """, (correo_personal,))
            
            docente = cursor.fetchone()
            
            if docente:
                # Convertir a dict para JSON
                docente_dict = dict(docente)
                
                return jsonify({
                    'success': True,
                    'found': True,
                    'docente': docente_dict
                }), 200
            else:
                return jsonify({
                    'success': False,
                    'found': False,
                    'message': 'No autorizado'
                }), 404
                
        except Exception as db_error:
            print(f"Error en base de datos: {db_error}")
            return jsonify({
                'success': False,
                'message': f'Error consultando base de datos: {str(db_error)}'
            }), 500
            
        finally:
            if cursor:
                cursor.close()
            if conn:
                conn.close()
                
    except Exception as e:
        print(f"Error en verify_teacher_email: {e}")
        return jsonify({
            'success': False,
            'message': f'Error interno: {str(e)}'
        }), 500
