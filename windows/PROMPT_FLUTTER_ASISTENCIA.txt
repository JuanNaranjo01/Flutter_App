// ⚠️ USA LA MISMA CONFIGURACIÓN HTTP QUE YA TIENES PARA REGISTER_STUDENT_EMBEDDINGS
// Solo cambia la URL del endpoint y los datos que envías

import 'dart:convert';
import 'package:camera/camera.dart';
import 'package:http/http.dart' as http;

// Función para registrar asistencia (similar a como registras embeddings)
Future<Map<String, dynamic>> registrarAsistencia(
    List<String> framesBase64) async {
  
  // LA MISMA BASE URL QUE YA USAS PARA REGISTER_STUDENT_EMBEDDINGS
  final url = Uri.parse(
      'http://192.168.100.99:5000/api/recognize_and_mark');
  
  try {
    // EL MISMO TIPO DE REQUEST QUE YA FUNCIONA EN TU APP
    final response = await http.post(
      url,
      headers: {'Content-Type': 'application/json'},
      body: jsonEncode({'frames': framesBase64}),
    ).timeout(Duration(seconds: 30));
    
    if (response.statusCode == 200) {
      return jsonDecode(response.body);
    } else {
      final error = jsonDecode(response.body);
      throw Exception(error['error'] ?? 'Error desconocido');
    }
  } catch (e) {
    throw Exception('Error de conexión: $e');
  }
}

// Ejemplo de captura de frames
Future<List<String>> capturar4Frames(CameraController controller) async {
  List<String> frames = [];
  
  for (int i = 0; i < 4; i++) {
    // Capturar imagen
    final XFile image = await controller.takePicture();
    
    // Convertir a base64
    final bytes = await image.readAsBytes();
    final base64Image = base64Encode(bytes);
    frames.add('data:image/jpeg;base64,$base64Image');
    
    // Esperar 500ms entre frames
    await Future.delayed(Duration(milliseconds: 500));
  }
  
  return frames;
}

// Ejemplo de uso completo
Future<void> iniciarRegistroAsistencia(
    CameraController controller) async {
  try {
    // Mostrar "Capturando..."
    print('Iniciando captura de frames...');
    
    // Capturar 4 frames
    final frames = await capturar4Frames(controller);
    
    // Mostrar "Analizando..."
    print('Enviando al servidor...');
    
    // Enviar al servidor
    final resultado = await registrarAsistencia(frames);
    
    // Mostrar resultado exitoso
    if (resultado['success'] == true) {
      final data = resultado['data'];
      print('✅ Asistencia registrada');
      print('Estudiante: ${data['estudiante']}');
      print('Materia: ${data['materia']}');
      print('Confianza: ${(data['confidence'] * 100).toStringAsFixed(1)}%');
    }
  } catch (e) {
    // Mostrar error
    print('❌ Error: $e');
  }
}
